\subsection{Requisiti hardware}
\begin{itemize}
  \item \textbf{Memoria RAM}:
    \begin{itemize}
      \item \emph{Minimo}: 4 GB
      \item \emph{Consigliato}: 8 GB o più (per prestazioni ottimali)
    \end{itemize}
  \item \textbf{Spazio su disco}: almeno 5 GB di spazio libero
  \item \textbf{Connessione Internet}: necessaria per l'installazione
\end{itemize}
