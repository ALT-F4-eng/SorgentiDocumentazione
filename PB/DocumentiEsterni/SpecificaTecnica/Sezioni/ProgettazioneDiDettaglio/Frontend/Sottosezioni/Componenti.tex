\subsubsection{Componenti}
\label{sec:Componenti}

\paragraph{TestListPage}
\label{sec:TestListPage}
\textbf{Elementi chiave}:
\begin{itemize}
    \item View Tests: Contenitore che mostra i test associati a un dataset;
    \item Scroll Bar: Barra di scorrimento verticale;
    \item Select Button: Pulsante per selezionare un test;
    \item Delete Button: Pulsante per eliminare un test;
    \item Compare Button: Pulsante che permette di confrontare i test e visualizzare la pagina di confronto;
    \item Search Bar: Barra di ricerca per filtrare i test;
    \item Upload Button: Pulsante per caricare un file di test;
    \item Rename Button: Pulsante per rinominare un test.
\end{itemize}

\paragraph{TestPage}
\textbf{Elementi chiave}:
\begin{itemize}
    \item View Test: Contenitore che mostra i dettagli di un test;
    \item Details list: lista contenente i seguenti campi:
    \begin{itemize}
        \item Name: Nome del test;
        \item ID: ID del test;
        \item Date: Data di creazione del test;
        \item LLM Model: Modello LLM utilizzato per il test;
        \item Dataset: Dataset utilizzato per il test;
        \item Responses: Risposte generate dal modello LLM;
        \item Values: Valori ottenuti dal test;
    \end{itemize}
    \item Scatterplot: Grafico di dispersione che mostra i risultati del test;
    \item Bar Chart: Grafico a barre che mostra i risultati del test suddivisi per range di valori;
    \item Save Button: Pulsante per salvare il test;
    \item Compare Button: Pulsante che permette di confrontare il test e visualizzare la pagina di confronto;
    \item Delete Button: Pulsante per eliminare il test;
    \item Rename Button: Pulsante per rinominare il test.
\end{itemize}

\paragraph{ComparisonPage}  
\textbf{Elementi chiave}:
\begin{itemize}
    \item View Comparison: Contenitore che mostra i dettagli di un confronto;
    \item Scatterplot Comparison: Grafico di dispersione per confrontare visivamente i risultati tra due o più test;
    \item Bar Chart Comparison: Grafico a barre che mostra i valori dei test comparati per range;
    \item Comparison Table: Tabella riepilogativa che confronta i dettagli dei test selezionati;
    \item Back Button: Pulsante per tornare alla pagina precedente;
    \item Legend: Legenda che aiuta a distinguere i test nei grafici.
\end{itemize}
