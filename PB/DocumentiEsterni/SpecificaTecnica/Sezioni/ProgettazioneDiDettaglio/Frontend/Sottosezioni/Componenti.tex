\subsection{Componenti}
\label{subsec:Componenti}

\subsubsection{LLMListPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item View LLMs: Contenitore per mostrare degli LLM presenti nel sistema.
    \item Scroll Bar: Bar per scorrere verticalmente.  
    \item Select Button: Pulsante per selezionare un LLM. 
    \item Config Button: Pulsante per aprire la configurazione della corrispondenza LLM.
    \item Delete Button: pulsante per cancellare la corrispondenza LLM
    \item Add Button: pulsante per andare alla pagina apposta per aggiungere LLM
\end{itemize}

\subsubsection{LLMDetailsPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item View Details: Contenitore che mostra tutte le informazioni a riguardo degli LLM selezionato
\end{itemize}
\subsubsection{LLMPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item Bodify Button: pulsante che abilità la modifica degli campi dati modificabili.
    \item Save Button: pulsatne che permette di salvare le modifiche fatte.
    \item Input Form: form dove l'utente può inserire le informazioni neccesari.
    \item Confirm Button: pulsate che permette l'utente di aggiungere LLM 
\end{itemize}
