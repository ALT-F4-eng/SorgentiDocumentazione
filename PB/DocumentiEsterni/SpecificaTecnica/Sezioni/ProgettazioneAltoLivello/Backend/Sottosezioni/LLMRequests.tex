\subsubsection{get\_llm\_datas}
\label{sec:get_LLM_datas}

\paragraph{Descrizione}
La funzione \texttt{get\_llm\_datas} recupera le infomrazioni relative all'LLM.

\paragraph{Dettagli dell'endpoint}
\begin{itemize}
    \item \textbf{HTTP Method}: GET;
    \item \textbf{Endpoint}: /llm/\{id\};
    \item \textbf{Response Model}: LLMDto;
    \item \textbf{Nome}: get\_llm\_datas;
    \item \textbf{Dependency Injection}:\\ \textbf{- LLM\_service(LLMUseCase)}: dipendenza risolta tramite \textit{Dependency Injector}.
    \item \textbf{Parametri}:\\ \textbf{- id(int)}: ID dell'LLM di cui recuperare le informazioni.
\end{itemize}

\paragraph{Implementazione}
\begin{itemize}
    \item Effettua una chiamata HTTP per recuperare le informazioni relative ad un LLM nello specifico.
    \item Restituisce un oggetto JSON con lo stato della risposta e, in caso di successo, i dati richiesti.
\end{itemize}

\subsubsection{write\_llm}
\label{sec:write_LLM}

\paragraph{Descrizione}
La funzione \texttt{write\_llm} salva o modifica un LLM.

\paragraph{Dettagli dell'endpoint}
\begin{itemize}
    \item \textbf{HTTP Method}: PUT;
    \item \textbf{Endpoint}: /llm/write;
    \item \textbf{Response Model}: LLMDto;
    \item \textbf{Nome}: write\_llm;
    \item \textbf{Dependency Injection}:\\ \textbf{- LLM\_service(LLMUseCase)}: dipendenza risolta tramite \textit{Dependency Injector}.
    \item \textbf{Parametri}:
    \begin{itemize}
        \item \textbf{- nome(str)}: nome dell'LLM.
        \item \textbf{- url(str)}: url dell'LLM.
        \item \textbf{- ka(str)}: chiave associata alla risposta dell'LLM.
        \item \textbf{- kq(str)}: chiave associata alla domanda dell'LLM.
        \item \textbf{- jh(dict)}: json di risposta alla creazione dell'header delle richieste HTTP verso l'LLM.
        \item \textbf{- jb(dict)}:json di risposta alla creazione del body delle richieste HTTP verso l'LLM.
    \end{itemize}
\end{itemize}

\paragraph{Implementazione}
\begin{itemize}
    \item Effettua una chiamata HTTP per il salvataggio o la modifica dell'LLM.
    \item Restituisce un oggetto JSON con lo stato della risposta e, in caso di successo, l'id dell'LLM.
\end{itemize}
