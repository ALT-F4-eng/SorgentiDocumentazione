\subsection{Componenti}
\label{subsec:Componenti}

\paragraph{StandardPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LogoIcon}: Un icona che rappresenta il logo del gruppo.
    \item \textbf{Menu}: Un insieme di link disponibili che permette all'utente di navigare tra le pagine diverse.
    \item \textbf{Footer}: Consente di visualizzare informazioni critiche.
    \item \textbf{LoadingScreen}: Mostra il grafico di caricamento.
    \item \textbf{MessageDialog}: Visualizzazione di un messaggio di finestra.
    \item \textbf{ErrorListener}: Verrà attivata la finestra del messaggio di errore nel caso in cui si verificasse un errore nel sistema;
    \item \textbf{ConfirmButton}: pulsante che permette di confermare le operazioni critiche.
\end{itemize}

\paragraph{LLMListPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMListContainer}: Contenitore che mostra i modelli LLM presenti nel sistema.
    \item \textbf{EmptyState}:  messaggio e grafica quando non ci sono LLM disponibili
\end{itemize}

\paragraph{LLMListView}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMListItem}: elemento della lista dei LLM salvati
    \item \textbf{AddButton}: pulsante per aggiungere un nuovo LLM; 
    \item \textbf{DoubleClickListener}: viene selezionato un LLM quando viene effettuato un doppio click su un elemento della lista. 
    \item \textbf{DeleteButton}: permette di cancellare il LLM corrispondente.
\end{itemize}
\textbf{Servizio} \\
\textbf{TestService}: servizio iniettato tramite `@Injectable`, fornisce le operazioni per la gestione dell'elenco LLM.

\paragraph{LLMView}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{ViewDetails}: mostrerà il nome, la data dell'ultima modifica, l'URL, informazioni sui header e body, e infine chiave per la domanda e la risposta sull'LLM selezionato.
    \item \textbf{ModifyButton}: pulsate per la modifica dell'LLM selezionato.
\end{itemize}
\textbf{TestService}: servizio iniettato tramite `@Injectable`, fornisce le informazioni del LLM selezionato.

\paragraph{LLMForm}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{NewNameLLM}: campo di input per modificare il nome dell'LLM;
    \item \textbf{NewURL}: campo di input per modificare L'URL del LLM;
    \item \textbf{NewKeyQuestion}: campo di input per modificare chiave associata alla domanda da porre all'LLM.
    \item \textbf{NewKeyAnswer}: campo di input per modificare chiave associata alla risposta da porre all'LLM.
    \item \textbf{NewKeyHeader} due campi di input per modificare ogni coppia chiave-valore associata alla creazione dell’header delle richieste HTTP verso l’LLM.
    \item \textbf{AddHeaderButton}: pulsante per l'inserimento di una oppia chiave-valore associata alla creazione dell’header delle richieste HTTP verso l’LLM.
    \item \textbf{NewKeyBody} due campi di input per modificare ogni coppia chiave-valore associata alla creazione del body delle richieste HTTP verso l’LLM
    \item \textbf{AddBodyButton}: pulsante per l'inserimento di una oppia chiave-valore associata alla creazione del body delle richieste HTTP verso l’LLM; 
    \item \textbf{SaveButton}: pulsante per salvare le modifiche;
    \item \textbf{AbortButton}: pulsante per annullare la modifica;
    \item \textbf{ValidationFeedback}: messaggi di errore per input vuoti o duplicati;
\end{itemize}
\textbf{TestService}: servizio iniettato tramite `@Injectable`, fornisce le informazioni del LLM selezionato.

\paragraph{LLMPage}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMContainer}: Contenitore che mostra le informazioni e le operazioni disponibili su LLM.
    \item \textbf{EmptyState}:  messaggio e grafica quando avviene un errore di caricamento.
\end{itemize}
