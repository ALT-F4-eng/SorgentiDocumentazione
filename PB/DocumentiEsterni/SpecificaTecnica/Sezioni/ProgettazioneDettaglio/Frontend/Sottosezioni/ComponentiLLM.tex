\subsubsection{LLMListPage}
\textbf{Componente padre}
\begin{itemize}
    \item StandardPage.
\end{itemize}
\textbf{Modulo}
\begin{itemize}
    \item LLMModule, modulo funzionale dell'applicazione che raccoglie e organizza tutti i componenti, servizi e risorse legati alla creazione, visualizzazione e gestione degli LLM.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMListContainer}: lista scorrevole degli LLM salvati, realizzata tramite sottocomponente \textbf{LLMListView};
    \item \textbf{AddButton}: pulsante che permette la creazione di un nuovo LLM, realizzata tramite sottocomponente \textbf{LLMForm};
    \item \textbf{DeleteMessage}: popup di conferma di eliminazione di un LLM, realizzato tramite il sottocomponente \textbf{ConfirmComponent};
    \item \textbf{ErrorMessage}: messaggio di errore visualizzato in seguito ad'un errore nell'ottenimento o nel salvataggio dei dati, realizzato tramite sottocomponente \textbf{MessageComponent}.
\end{itemize}
\textbf{Servizio}
\begin{itemize}
    \item \textbf{LLMService}: servizio iniettato tramite `@Injectable`, fornisce le operazioni per la gestione di un elemento della lista di LLM.
\end{itemize}
\subsubsection{LLMListView}
\textbf{Componente padre}
\begin{itemize}
    \item LLMListPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMList}: lista scorrevole degli LLM salvati, permette la creazione di un sottocomponente \textbf{LLMElement} per ogni LLM salvato (tramite la direttiva *\texttt{ngFor});
    \item \textbf{StatusMessage}: visualizzato un messaggio che indica all'utente nel caso non ci siano LLM salvati nel sistema;
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item LLMs: lista contenente tutti i LLM salvati nel sistema.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item viewLLM: evento emesso quando riceve da \textbf{LLMElement} la richiesta di visualizzare un LLM;
    \item deleteLLM: evento emesso quando riceve da \textbf{LLMElement} la richiesta di eliminazione di un LLM;
\end{itemize}
\subsubsection{LLMElement}
\textbf{Componente padre}
\begin{itemize}
    \item LLMListView.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMNameText}: sezione testuale contenente il nome dell'LLM;
    \item \textbf{LLMDateText}: sezione testuale contenente la data dell'ultima modifica dell'LLM;
    \item \textbf{loadButton}: pulsante per richiedere il caricamento della pagina di visualizzazione del LLM;
    \item \textbf{DeleteButton}: pulsante per richiedere l'eliminazione dell'LLM, questa viene poi confermata o annullata tramite il sottocomponente \textbf{ConfirmComponent};
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item LLM: LLM associato al componente.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item load: evento emesso quando viene premuto il pulsante per richiedere visualizzazione dell'LLM.
    \item delete: evento emesso quando viene premuto il pulsante per richiedere eliminazione dell'LLM;
\end{itemize}
\subsubsection{LLMPage}
\textbf{Componente padre}
\begin{itemize}
    \item StandardPage.
\end{itemize}
\textbf{Modulo}
\begin{itemize}
    \item LLMModule, modulo funzionale dell'applicazione che raccoglie e organizza tutti i componenti, servizi e risorse legati alla creazione, visualizzazione e gestione degli LLM.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LLMContainer}: visualizza le informazioni dettagliate del LLM selezionato, realizzato tramite sottocomponente \textbf{LLMPageView}

\end{itemize}
\textbf{Servizio}
\begin{itemize}

\end{itemize}
\subsubsection{LLMPageView}
\textbf{Componente padre}
\begin{itemize}
    \item LLMPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{NameText}: sezione testuale contenente il nome del LLM; 
    \item \textbf{LastUpdateText}: sezione testuale contenente la data dell'ultima modifica del LLM;
    \item \textbf{URLText}: sezione testuale contenente l'URL usato per l'esecuzione delle chiamate HTTP all'API dell'LLM;
    \item \textbf{HeaderText}: sezione testuale contenente le coppie chiave-valore per la costruzione dell'header delle chiamate;
    \item \textbf{BodyText}: sezione testuale contenente le coppie chiave-valore per la costruzione del body delle chiamate;
    \item \textbf{QuestionText}: sezione testuale contenente la chiave da utilizzare per specificare la domanda da porre all'LLM nel body delle richieste HTTP;
    \item \textbf{AnswerText}: sezione testuale contenente la chiave da utilizzare per estrarre la risposta data dall'LLM contenuta nelle risposte HTTP;
    \item \textbf{ModifyButton}: pulsate per abilitare la modifica dell'LLM selezionato.
\end{itemize}
\textbf{input}
\begin{itemize}
    \item name: riceve le informazioni del LLM selezionato.
    \item date: data dell'ultima modifica del LLM;
    \item url: URL usato per l'esecuzione delle chiamate HTTP all'API dell'LLM;
    \item header: coppie chiave-valore per la costruzione dell'header delle chiamate;
    \item body: coppie chiave-valore per la costruzione del body delle chiamate;
    \item question: chiave da utilizzare per specificare la domanda da porre all'LLM nel body delle richieste HTTP;
    \item answer: chiave da utilizzare per estrarre la risposta data dall'LLM contenuta nelle risposte HTTP;
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item modifyLLM: evento emesso quando viene premuto il pulsante per richiedere la modifica dell'LLM;
\end{itemize}
\subsubsection{LLMForm}
\textbf{Componente padre}
\begin{itemize}
    \item LLMListPage;
    \item LLMPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{NameLabel}: etichetta associata al campo di inserimento del nome;
    \item \textbf{NameInput}: campo di input per modificare il nome dell'LLM;
    \item \textbf{URLLabel}: etichetta associata al campo di inserimento del URL;
    \item \textbf{URLInput}: campo di input per modificare L'URL del LLM;
    \item \textbf{KeyHeaderLabel}: etichetta associata al campo di inserimento della nuova chiave dell'header;
    \item \textbf{KeyHeaderInput}: campo di input per modificare la nuova chiave dell'header;
    \item \textbf{ValueHeaderLabel}: etichetta associata al campo di inserimento del nuovo valore dell'header;
    \item \textbf{ValueHeaderInput}: campo di input per modificare il nuovo valore dell'header;
    \item \textbf{AddHeaderButton}: pulsante per aggiungere la coppia chiave-valore all'header;
    \item \textbf{KeyBodyLabel}: etichetta associata al campo di inserimento della nuova chiave del body;
    \item \textbf{KeyBodyInput}: campo di input per modificare la nuova chiave del body;
    \item \textbf{ValueBodyLabel}: etichetta associata al campo di inserimento del nuovo valore del body;
    \item \textbf{ValueBodyInput}: campo di input per modificare il nuovo valore del body;
    \item \textbf{AddBodyButton}: pulsante per aggiungere la coppia chiave-valore al body;
    \item \textbf{QuestionKeyLabel}: etichetta associata al campo di inserimento della chiave da associare alle domande poste all'LLM;
    \item \textbf{QuestionKeyInput}: campo di input per modificare la chiave da associare alle domande poste all'LLM;
    \item \textbf{AnswerKeyLabel}: etichetta associata al campo di inserimento della chiave da associare alle risposte fornite dall'LLM;
    \item \textbf{AnswerKeyInput}: campo di input per modificare la chiave da associare alle risposte fornite dall'LLM;
    \item \textbf{SaveButton}: pulsante per salvare le modifiche;
    \item \textbf{AbortButton}: pulsante per annullare la modifica;
    \item \textbf{ErrorMessage}: messaggio di errore visualizzato in seguito alla compilazione errata di una coppia chiave-valore, realizzato tramite sottocomponente \textbf{MessageComponent};
    \item \textbf{StatusMessage}: popup di conferma di salvataggio delle modifiche, realizzato tramite sottocomponente \textbf{MessageComponent};
\end{itemize}
\textbf{input}
\begin{itemize}
    \item name: nome del LLM da modificare se presente; 
    \item url: URL del LLM da modificare;
    \item question: chiave da associare alle domande poste all'LLM;
    \item answer: chiave da associare alle risposte fornite dall'LLM;
    \item headerKeys: chiavi associate all'header delle chiamate;
    \item headerValues: valori associati all'header delle chiamate;
    \item bodyKeys: chiavi associate al body delle chiamate;
    \item bodyValues: valori associati al body delle chiamate;
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item addLLM: evento emesso quando viene confermata l'aggiunta di un LLM;
    \item modifyLLM: evento emesso quando viene confermata la modifica di un LLM.
\end{itemize}
