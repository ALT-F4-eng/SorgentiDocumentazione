\subsection{Componenti}
\label{subsec:Componenti}

\subsubsection{StandardPage}
\textbf{Modulo}
\begin{itemize}
    \item StandardModule, modulo funzionale dell'applicazione, ogni pagina devono averla.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{LogoIcon}: Un icona che rappresenta il logo del gruppo.
    \item \textbf{Menu}: Un insieme di link disponibili che permette all'utente di navigare tra le pagine diverse.
    \item \textbf{Footer}: Consente di visualizzare informazioni critiche.
    \item \textbf{LoadingScreen}: Mostra il grafico di caricamento.
    \item \textbf{MessageDialog}: Visualizzazione di un messaggio di finestra.
    \item \textbf{ErrorListener}: Verrà attivata la finestra del messaggio di errore nel caso in cui si verificasse un errore nel sistema;
    \item \textbf{ConfirmButton}: pulsante che permette di confermare le operazioni critiche.
\end{itemize}

\subsubsection{HomePage}
\textbf{Modulo}
\begin{itemize}
    \item HomeModule, modulo funzionale dell'applicazione mostra le informazioni riguardo all'applicazione.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{ContentText}:sezione testuale contenente le informazioni sull'applicazione.
\end{itemize}
