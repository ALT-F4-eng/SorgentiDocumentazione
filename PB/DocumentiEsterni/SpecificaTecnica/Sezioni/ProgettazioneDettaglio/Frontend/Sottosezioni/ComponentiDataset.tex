\subsubsection{Componenti}
\label{subsubsec:componenti}
\paragraph{DatasetListPage}
\textbf{Modulo}
\begin{itemize}
    \item DatasetModule, modulo funzionale dell'applicazione che raccoglie e organizza tutti i componenti, servizi e risorse legati alla creazione, visualizzazione e gestione di dataset.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{SearchDataset}: barra per la ricerca dei dataset, realizzata tramite sottocomponente \textbf{SearchBar};
    \item \textbf{DatasetListContainer}: lista scorrevole dei dataset salvati, realizzata tramite sottocomponente \textbf{DatasetListView};
    \item \textbf{CreateDatasetButton}: pulsante che permette la creazione di un dataset temporaneo;
    \item \textbf{CreateDatasetFromJSONButton}: pulsante che permette di richiedere la creazione di un dataset a partire da un file JSON, la creazione viene poi realizzata tramite \textbf{DatasetJSONForm};
    \item \textbf{StatusMessage}: popup di conferma di eliminazione di un dataset o di sovrascrittura del dataset caricato (se modificato), realizzato tramite sottocomponente \textbf{ConfirmComponent};
    \item \textbf{ErrorMessage}: messaggio di errore visualizzato in seguito a un errore nell'ottenimento o nel salvataggio dei dati, realizzato tramite sottocomponente \textbf{MessageComponent}.
\end{itemize}
\textbf{Servizio}
\begin{itemize}
    \item DatasetService, servizio che permette l'esecuzione di operazioni CRUD sulla lista dei dataset salvati.
\end{itemize}

\paragraph{DatasetListView}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetListPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{StatusMessage}: paragrafo contenente il messaggio che viene visualizzato dall'utente nel caso non ci siano dataset salvati nel sistema;
    \item \textbf{DatasetList}: lista scorrevole dei dataset salvati nel sistema, permette la creazione di un sottocomponente \textbf{DatasetElement} per ogni dataset salvato (tramite la direttiva \texttt{*ngFor}).
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item data sets: lista contenente tutti i dataset salvati e/o ricercati nel sistema.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item rename Dataset: evento emesso quando riceve da \textbf{DatasetElement} la richiesta di rinomina di un dataset;
    \item copy Dataset: evento emesso quando riceve da \textbf{DatasetElement} la richiesta di copia di un dataset;
    \item delete Dataset: evento emesso quando riceve da \textbf{DatasetElement} la richiesta di eliminazione di un dataset;
    \item load Dataset: evento emesso quando riceve da \textbf{DatasetElement} la richiesta di caricare un dataset.
\end{itemize}

\paragraph{DatasetElement}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetListView.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{NameText}: sezione testuale contenente il nome del dataset;
    \item \textbf{DatasetDate}: sezione testuale contenente la data dell'ultima modifica del dataset;
    \item \textbf{RenameButton}: pulsante per richiedere la rinominazione del dataset, questa viene poi realizzata tramite il sottocomponente \textbf{DatasetNameForm};
    \item \textbf{CopyButton}: pulsante per richiedere la copia del dataset;
    \item \textbf{DeleteButton}: pulsante per richiedere l'eliminazione del dataset, questa viene poi confermata o annullata tramite il sottocomponente \textbf{ConfirmComponent};
    \item \textbf{LoadButton}: pulsante per richiedere il caricamento del dataset, se il dataset attualmente caricato presenta modifiche viene richiesta la conferma o l'annullamento della sovrascrittura tramite il sottocomponente \textbf{ConfirmComponent}.
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item dataset: dataset associato al componente.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item rename: evento emesso quando viene premuto il pulsante per richiedere la rinominazione del dataset;
    \item copy: evento emesso quando viene premuto il pulsante per richiedere la copia del dataset;
    \item delete: evento emesso quando viene premuto il pulsante per richiedere l'eliminazione del dataset;
    \item load: evento emesso quando viene premuto il pulsante per richiedere il caricamento del dataset.
\end{itemize}

\paragraph{DatasetNameForm}

\begin{itemize}

    \item \textbf{NewNameInput}: campo di inserimento del nuovo nome da assegnare al dataset;
    \item \textbf{ConfirmRenameButton}: pulsante che permette di confermare la richiesta di assegnazione o modifica del nome del dataset;
    \item \textbf{CancelRenameButton}: pulsante che permette di annullare la richiesta di assegnazione o modifica del nome del dataset.
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item currentName: gli viene assegnato il nome del dataset nel caso non sia temporaneo.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item confirmRename: evento emesso quando viene premuto il pulsante per confermare l'assegnazione o modifica del nome del dataset;
    \item cancelRename: evento emesso quando viene premuto il pulsante per annullare l'assegnazione o modifica del nome del dataset. 
\end{itemize}

\paragraph{DatasetContentPage}
\textbf{Modulo}
\begin{itemize}
    \item DatasetModule, modulo funzionale dell'applicazione che raccoglie e organizza tutti i componenti, servizi e risorse legati alla creazione, visualizzazione e gestione di dataset.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{NameText}: sezione testuale che permette di visualizzare il nome del dataset caricato, nel caso non sia temporaneo;
    \item \textbf{TemporaryLabel}: sezione testuale che indica se il dataset caricato è temporaneo;
    \item \textbf{TestButton}: pulsante che permette l'esecuzione di un test sul dataset caricato (se non è vuoto), l'LLM da testare viene selezionato successivamente tramite il sottocomponente \textbf{LLMSelectionList};
    \item \textbf{SaveButton}: pulsante che permette il salvataggio delle modifiche non salvate realizzate sul dataset caricato, se il dataset caricato è temporaneo viene successivamente richiesto di assegnargli un nome tramite il sottocomponente \textbf{DatasetNameForm};
    \item \textbf{SearchQA}: barra per la ricerca di coppie di domanda e risposta nel dataset, realizzata tramite sottocomponente \textbf{SearchBar}; 
    \item \textbf{QAContainer}: sezione di visualizzazione delle coppie domanda e risposta del dataset caricato, realizzata dal sottocomponente \textbf{DatasetPageView}.
    \item \textbf{QAForm}: form che permette l'inserimento e la modifica di una coppia domanda e risposta del dataset caricato, realizzato tramite il sottocomponente \textbf{QAForm};
    \item \textbf{ErrorMessage}: messaggio di errore visualizzato in seguito a un errore nell'ottenimento o nel salvataggio dei dati o nell'esecuzione di un test, realizzato tramite sottocomponente \textbf{MessageComponent};
    \item \textbf{UnfilledErrorMessage}: messaggio di errore visualizzato nel caso si richieda un test su un dataset con elementi incompleti, indicati nel messaggio tramite link, realizzato tramite sottocomponente \textbf{MessageComponent}.
\end{itemize}
\textbf{Servizio}
\begin{itemize}
    \item QAService, servizio che permette l'esecuzione di operazioni CRUD sulla lista delle coppie domanda e risposta di un dataset;
    \item TestService, servizio che permette l'esecuzione del test sul dataset caricato.
\end{itemize}

\paragraph{DatasetPageView}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetContentPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{QAListContainer}: pagina delle coppie domanda e risposta del dataset caricato, realizzata tramite il sottocomponente \textbf{QAListView}.
    \item \textbf{QANavBar}: sezione che permette la navigazione tra le pagine del dataset, realizzata tramite sottocomponente \textbf{PageNavigation}.
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item dataset: dataset associato al componente;
    \item QAlist: elenco di coppie domanda e risposta relative alla pagina di visualizzazione del dataset associato.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item changePage: evento emesso quando riceve da \textbf{PageNavigation} la richiesta di cambio pagina;
    \item editQA: evento emesso quando riceve da \textbf{QAListView} la richiesta di modifica di una coppia domanda e risposta nella lista;
    \item deleteQA: evento emesso quando riceve da \textbf{QAListView} la richiesta di eliminazione di una coppia domanda e risposta nella lista;
\end{itemize}

\paragraph{QAListView}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetPageView.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{QAList}: lista scorrevole delle coppie domanda e risposta del dataset caricato appartenenti a una pagina, permette la creazione di un sottocomponente \textbf{QAListElement} per ogni coppia domanda e risposta della pagina (tramite la direttiva \texttt{*ngFor}).
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item QAlist: elenco di coppie domanda e risposta relative alla pagina di visualizzazione del dataset associato.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item requestEditQAElement: evento emesso quando riceve da \textbf{QAListElement} la richiesta di modifica della coppia domanda e risposta associata;
    \item requestDeleteQAElement: evento emesso quando riceve da \textbf{QAListElement} la conferma dell'eliminazione della coppia domanda e risposta associata.
\end{itemize}

\paragraph{QAListElement}
\textbf{Componente padre}
\begin{itemize}
    \item QAListView.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{QuestionText}: sezione testuale che permette di visualizzare la domanda della coppia domanda e risposta associata al componente;
    \item \textbf{AnswerText}: sezione testuale che permette di visualizzare la risposta della coppia domanda e risposta associata al componente;
    \item \textbf{MessageText}: sezione testuale che indica se la domanda o la risposta sono vuote;
    \item \textbf{EditButton}: pulsante che permette di richiedere la modifica della coppia domanda e risposta associata al componente;
    \item \textbf{DeleteButton}: pulsante che permette di richiedere l'eliminazione della coppia domanda e risposta associata al componente;
    \item \textbf{ConfirmWindow}: popup che permette di confermare o annullare la richiesta di eliminazione della coppia domanda e risposta associata al componente.
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item QAelement: coppia domanda e risposta associata al componente.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item editQAElement: evento emesso quando viene premuto il pulsante per richiedere la modifica della coppia domanda e risposta;
    \item deleteQAElement: evento emesso quando viene confermata la richiesta di eliminazione della coppia domanda e risposta.
\end{itemize}

\paragraph{QAForm}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetContentPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{QuestionLabel}: etichetta associata al campo di inserimento della nuova domanda;
    \item \textbf{QuestionInput}: campo di inserimento della domanda per la nuova coppia domanda e risposta da aggiungere, o che permette di modificare la domanda della coppia che si vuole modificare;
    \item \textbf{AnswerLabel}: etichetta associata al campo di inserimento della nuova risposta;
    \item \textbf{AnswerInput}: campo di inserimento che permette di inserire la risposta per la nuova coppia domanda e risposta da aggiungere, o che permette di modificare la risposta della coppia che si vuole modificare;
    \item \textbf{ConfirmButton}: pulsante che permette di confermare l'inserimento o la modifica della coppia domanda e risposta;
    \item \textbf{CancelButton}: pulsante che permette di annullare l'inserimento o la modifica della coppia domanda e risposta; 
\end{itemize}
\textbf{Input}
\begin{itemize}
    \item QAelement: gli viene fornita la coppia domanda e risposta nel caso ne venga richiesta la modifica;
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item confirmEditQA: evento emesso quando viene premuto il pulsante per confermare l'inserimento o la modifica della coppia domanda e risposta;
    \item cancelEditQA: evento emesso quando viene premuto il pulsante per annullare l'inserimento o la modifica della coppia domanda e risposta.
\end{itemize}

\paragraph{LLMSelectionList}
\textbf{Componente padre}
\begin{itemize}
    \item DatasetContentPage.
\end{itemize}
\textbf{Elementi chiave}
\begin{itemize}
    \item \textbf{SelectLLM}: lista di elementi rappresentanti LLM, ogni LLM è indicato con nome e data di ultima modifica, ogni LLM è selezionabile.
\end{itemize}
\textbf{Output}
\begin{itemize}
    \item runLLMTest: evento emesso quando viene selezionato un LLM della lista per utilizzarlo nel test.
\end{itemize}
\textbf{Servizio}
\begin{itemize}
    \item LLMService: servizio utilizzato per ottenere la lista degli LLM salvati sui quali è possibile effettuare il test.
\end{itemize}
